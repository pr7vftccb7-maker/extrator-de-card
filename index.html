function copiarAssinadasDoLote(idLote) { const lote = appState.emails.lotes.find(l => l.idLote === idLote); if (!lote) return; const itensAssinadosParaCopiar = lote.itens.filter(item => { const emailChave = item.principal.toLowerCase(); const idUnico = `item_emails_${lote.idLote}_${item.idItem}`; const statusAtual = appState.emails.status[emailChave] || appState.emails.status[idUnico] || item.status || ''; const jaFoiCopiadoAssinada = appState.emails.usados.includes(idUnico); return statusAtual === 'assinada' && !jaFoiCopiadoAssinada; }); if (itensAssinadosParaCopiar.length === 0) { alert('⚠️ Nenhuma conta "assinada" nova encontrada neste lote.'); return; } let textoParaClipboard = ''; itensAssinadosParaCopiar.forEach(item => { textoParaClipboard += item.textoCompleto + '\n'; const idUnico = `item_emails_${lote.idLote}_${item.idItem}`; if (!appState.emails.usados.includes(idUnico)) { appState.emails.usados.push(idUnico); } }); copiarParaAreaTransferencia(textoParaClipboard.trim()).then(() => { localStorage.setItem(CHAVES.emails.usados, JSON.stringify(appState.emails.usados)); // CRIA UM NOVO LOTE SEPARADO PARA CADA CÓPIA COM DATA E HORA const itensNovosLote = itensAssinadosParaCopiar.map((item, idx) => ({ idItem: idx + 1, principal: item.principal, secundario: item.secundario, terciario: '', textoCompleto: item.textoCompleto })); const novoLoteAssinadas = { idLote: Date.now(), nome: `ASSINADAS • ${gerarDataHoraFormatada()}`, banner: BANNERS[appState.emails.lotes.length % BANNERS.length], itens: itensNovosLote }; appState.emails.lotes.unshift(novoLoteAssinadas); reordenarEAtualizarLotes('emails'); localStorage.setItem(CHAVES.emails.lotes, JSON.stringify(appState.emails.lotes)); renderizarTodosLotes('emails'); mostrarStatus(`✅ Novo lote criado: ASSINADAS • ${gerarDataHoraFormatada()}`); }).catch(() => alert('❌ Erro ao acessar a área de transferência.')); }
